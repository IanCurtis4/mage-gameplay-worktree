## Diagnostic (Sol, checkpoint36): same walk with background navigation on; long frames and click->route latency.
const Launcher = preload("res://scripts/map_authoring/preview/gameplay_launcher.gd")
const App = preload("res://scripts/map_authoring/preview/gameplay_app.gd")
const View = preload("res://scripts/world/expedition_geometry_view.gd")
func _initialize() -> void: run.call_deferred()
func run() -> void:
	var path := ""
	for arg: String in OS.get_cmdline_user_args():
		if arg.begins_with("--candidate="): path = arg.trim_prefix("--candidate=")
	var candidate: Variant = bytes_to_var(FileAccess.get_file_as_bytes(path))
	var created := Launcher.create_session(candidate)
	var window := Launcher.create_window(created.path) if created.ok else {"ok":false}
	var app := App.new(); app.quit_on_close = false; app.hold_start = true; root.add_child(app); current_scene = app
	var opened: Dictionary = await app.open_window_async(window.path)
	var started: Dictionary = await app._start_with_gate() if opened.ok else {"ok":false}
	if not started.ok: print("WP start_failed "+JSON.stringify({"created":created.get("ok",false),"created_code":String(created.get("error_code",created.get("code",""))),"window":window.get("ok",false),"window_code":String(window.get("error_code",window.get("code",""))),"opened":opened,"started":started})); quit(1); return
	var main = app.host; var player: PlayerActor = main.player
	for e: CombatActor in main.enemies: e.process_mode = Node.PROCESS_MODE_DISABLED  # probe only: no combat
	app.begin()
	main.expedition_view.geometry.navigation_async = true
	var start_pos: Vector3 = player.world_point().position
	var goal := Vector3(1640,0,840)
	var steps := 14
	for i: int in range(1,steps+1):
		var target := start_pos.lerp(goal,float(i)/steps)
		var clicked: Dictionary = main.spatial_pointer_move(View.project(target))
		var t0 := Time.get_ticks_msec(); var last := Time.get_ticks_usec(); var worst := 0.0
		var routed := -1
		while Time.get_ticks_msec()-t0 < 8000 and (player.world_motion.has_route() or not main._pending_world_click.is_empty()):
			await process_frame
			var dt := (Time.get_ticks_usec()-last)/1000.0; last = Time.get_ticks_usec()
			if dt > 400.0: print("WP long_frame "+JSON.stringify({"step":i,"dt_ms":snappedf(dt,0.1),"pos":str(player.world_point().position),"surface":player.world_point().surface_id}))
			worst = maxf(worst,dt)
			if routed < 0 and player.world_motion.has_route(): routed = Time.get_ticks_msec()-t0
		print("WP step "+JSON.stringify({"i":i,"click_ok":clicked.get("ok",false),"code":String(clicked.get("error_code","")),"pos":str(player.world_point().position),"surface":player.world_point().surface_id,"worst_ms":snappedf(worst,0.1),"pending":clicked.get("pending",false),"route_after_ms":routed,"ms":Time.get_ticks_msec()-t0}))
	print("WP_DONE"); quit(0)
