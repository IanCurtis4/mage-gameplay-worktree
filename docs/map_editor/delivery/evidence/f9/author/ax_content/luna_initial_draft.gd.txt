func find_placement(source: Dictionary, placement_id: String) -> Dictionary:
	for placement: Dictionary in source.get("placements",[]):
		if placement.placement_id == placement_id: return placement
	return {}

func find_record(records: Array, key: String, id: String) -> Dictionary:
	for record: Dictionary in records:
		if str(record.get(key,"")) == id: return record
	return {}

func find_record_index(records: Array, key: String, id: String) -> int:
	for index: int in records.size():
		if str(records[index].get(key,"")) == id: return index
	return -1

func find_script(parent: Node, script: Script) -> Variant:
	if parent.get_script() == script: return parent
	for child: Node in parent.get_children():
		var found: Variant = find_script(child,script)
		if found != null: return found
	return null

func finish() -> void:
	Presentation.active_mode = "ux"

	for shell: Variant in [gui_shell,ax_shell]:
		if is_instance_valid(shell): shell.free()
	print("F9_AX_CONTENT_RESULT "+JSON.stringify({"ok":ok,"count":checks.size(),"expected_count":EXPECTED_COUNT,"checks":checks,"source_hashes":hashes}))
	quit(0 if ok else 1)
