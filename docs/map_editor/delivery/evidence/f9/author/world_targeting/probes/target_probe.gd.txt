	if not opened.ok: print("TARGET_PROBE open "+JSON.stringify(opened)); quit(1); return

	view.sync()
	print("TARGET_PROBE view_transform "+str(view.global_transform)+" canvas "+str(main.get_canvas_transform()))

		if not is_instance_valid(enemy) or enemy.world_motion == null: continue
		var wp: ExpeditionWorldPoint = enemy.world_point()

		var foot: Vector2 = view.to_global(foot_local)
		var vis: Dictionary = view.pick(foot,view.geometry.geometry_revision)
		var scale := enemy.sprite_visual_scale
		var hits := 0; var samples := 0; var hit_cells: Array = []

				var probe := enemy.global_position+Vector2(dx,-dy)

			"visible_surface":String(vis.get("data",{}).get("point",{}).get("surface_id","")) if vis.get("ok",false) else "",

	for row: Dictionary in rows:

		print("TARGET_PROBE enemy "+JSON.stringify(row))
